cmake_minimum_required(VERSION 3.22)
project(dlstreamer_selected LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(PkgConfig REQUIRED)
find_package(OpenVINO REQUIRED COMPONENTS Runtime)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.28.2 gstreamer-base-1.0 gstreamer-video-1.0 gstreamer-analytics-1.0 gstreamer-allocators-1.0 gstreamer-rtp-1.0 gstreamer-va-1.0)
pkg_check_modules(VA REQUIRED IMPORTED_TARGET libva libva-drm)
set(DLS_SOURCE "" CACHE PATH "DL Streamer v2026.2.0 source")
set(SYSROOT "" CACHE PATH "Rootless Ubuntu development packages")
if(NOT EXISTS "${DLS_SOURCE}/src/monolithic/gst/elements/gvatrack/tracker.cpp")
  message(FATAL_ERROR "Set DLS_SOURCE to the patched v2026.2.0 checkout")
endif()
set(mono "${DLS_SOURCE}/src/monolithic")
set(track "${mono}/gst/elements/gvatrack")
set(convert "${mono}/gst/elements/gvametaconvert")
set(publish "${mono}/gst/elements/gvametapublish")
file(WRITE "${CMAKE_BINARY_DIR}/config.h" "#pragma once\n#define ENABLE_VAAPI\n#define PLUGIN_VERSION \"2026.2.0-amd-selected\"\n#define PLUGIN_LICENSE \"MIT/X11\"\n#define PACKAGE \"dlstreamer\"\n#define PACKAGE_NAME \"DL Streamer selected elements\"\n#define PRODUCT_FULL_NAME \"DL Streamer\"\n#define GST_PACKAGE_ORIGIN \"https://github.com/open-edge-platform/dlstreamer\"\n")
file(WRITE "${CMAKE_BINARY_DIR}/gvametapublish_export.h" "#pragma once\n#define GVAMETAPUBLISH_EXPORTS\n")
file(GLOB_RECURSE tracker_sources "${track}/*.cpp" "${track}/*.c")
file(GLOB convert_sources "${convert}/*.cpp" "${convert}/*.c")
file(GLOB meta_sources "${DLS_SOURCE}/src/gst/metadata/*.c")
file(GLOB utils_sources "${DLS_SOURCE}/src/utils/*.cpp")
add_library(amdva_dlstreamer_common SHARED ${meta_sources} ${utils_sources}
  "${mono}/gst/common/gva_caps.c" "${mono}/gst/common/gva_utils.cpp" "${mono}/gst/common/logger_functions.cpp"
  "${mono}/inference_backend/logger/logger.cpp" "${mono}/inference_backend/logger/perf_logger.cpp")
add_library(gstdlstreamer_selected MODULE register.cpp ${tracker_sources} ${convert_sources})
file(GLOB publish_sources "${publish}/*.cpp" "${publish}/base/*.cpp" "${publish}/file/*.cpp")
add_library(gstgvametapublish MODULE ${publish_sources})
foreach(target amdva_dlstreamer_common gstdlstreamer_selected gstgvametapublish)
  target_include_directories(${target} PRIVATE
    "${CMAKE_BINARY_DIR}" "${DLS_SOURCE}/include" "${DLS_SOURCE}/include/dlstreamer/gst/metadata"
    "${DLS_SOURCE}/include/dlstreamer/gst/videoanalytics" "${DLS_SOURCE}/src/utils" "${mono}/gst/common"
    "${mono}/inference_backend/include" "${track}" "${convert}" "${publish}" "${publish}/base"
    "${SYSROOT}/usr/include/opencv4" "${SYSROOT}/usr/include")
  target_link_directories(${target} PRIVATE "${SYSROOT}/usr/lib/x86_64-linux-gnu")
  target_link_libraries(${target} PRIVATE PkgConfig::GST PkgConfig::VA openvino::runtime opencv_core opencv_imgproc ${CMAKE_DL_LIBS})
  target_compile_definitions(${target} PRIVATE GST_USE_UNSTABLE_API)
  target_link_options(${target} PRIVATE -Wl,--no-undefined)
endforeach()
target_link_libraries(gstdlstreamer_selected PRIVATE amdva_dlstreamer_common)
target_link_libraries(gstgvametapublish PRIVATE amdva_dlstreamer_common)
install(TARGETS amdva_dlstreamer_common LIBRARY DESTINATION lib)
install(TARGETS gstdlstreamer_selected gstgvametapublish LIBRARY DESTINATION lib/gstreamer-1.0)
