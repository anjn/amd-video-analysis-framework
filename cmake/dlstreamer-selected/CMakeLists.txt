cmake_minimum_required(VERSION 3.22)
project(dlstreamer_selected LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(PkgConfig REQUIRED)
find_package(OpenVINO REQUIRED COMPONENTS Runtime)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.28.2 gstreamer-base-1.0 gstreamer-video-1.0 gstreamer-analytics-1.0 gstreamer-allocators-1.0 gstreamer-rtp-1.0 gstreamer-va-1.0)
pkg_check_modules(VA REQUIRED IMPORTED_TARGET libva libva-drm)
set(DLS_SOURCE "" CACHE PATH "DL Streamer v2026.2.0 source")
set(SYSROOT "" CACHE PATH "Rootless Ubuntu development packages")
if(NOT EXISTS "${DLS_SOURCE}/src/monolithic/gst/elements/gvatrack/tracker.cpp")
  message(FATAL_ERROR "Set DLS_SOURCE to the patched v2026.2.0 checkout")
endif()
set(mono "${DLS_SOURCE}/src/monolithic")
set(track "${mono}/gst/elements/gvatrack")
set(convert "${mono}/gst/elements/gvametaconvert")
set(publish "${mono}/gst/elements/gvametapublish")
file(WRITE "${CMAKE_BINARY_DIR}/config.h" "#pragma once\n#define ENABLE_VAAPI\n#define PLUGIN_VERSION \"2026.2.0-amd-selected\"\n#define PLUGIN_LICENSE \"MIT/X11\"\n#define PACKAGE \"dlstreamer\"\n#define PACKAGE_NAME \"DL Streamer selected elements\"\n#define PRODUCT_FULL_NAME \"DL Streamer\"\n#define GST_PACKAGE_ORIGIN \"https://github.com/open-edge-platform/dlstreamer\"\n")
file(WRITE "${CMAKE_BINARY_DIR}/gvametapublish_export.h" "#pragma once\n#define GVAMETAPUBLISH_EXPORTS\n")
file(GLOB_RECURSE tracker_sources "${track}/*.cpp" "${track}/*.c")
file(GLOB convert_sources "${convert}/*.cpp" "${convert}/*.c")
file(GLOB meta_sources "${DLS_SOURCE}/src/gst/metadata/*.c")
file(GLOB utils_sources "${DLS_SOURCE}/src/utils/*.cpp")
add_library(amdva_dlstreamer_common SHARED ${meta_sources} ${utils_sources}
  "${mono}/gst/common/gva_caps.c" "${mono}/gst/common/gva_utils.cpp" "${mono}/gst/common/logger_functions.cpp"
  "${mono}/inference_backend/logger/logger.cpp" "${mono}/inference_backend/logger/perf_logger.cpp")
set(extra_sources)
set(extra_includes)
foreach(element gvastreammux gvastreamdemux gvametaaggregate gvaattachroi gvaanalytics gvafpscounter gvafpsthrottle)
  file(GLOB element_sources "${mono}/gst/elements/${element}/*.cpp" "${mono}/gst/elements/${element}/*.c")
  list(FILTER element_sources EXCLUDE REGEX "/register_[^/]+\\.c$")
  list(APPEND extra_sources ${element_sources})
  list(APPEND extra_includes "${mono}/gst/elements/${element}")
endforeach()
add_library(gstdlstreamer_selected MODULE register.cpp ${tracker_sources} ${convert_sources} ${extra_sources})
target_include_directories(gstdlstreamer_selected PRIVATE ${extra_includes})
file(GLOB publish_sources "${publish}/*.cpp" "${publish}/base/*.cpp" "${publish}/file/*.cpp")
add_library(gstgvametapublish SHARED ${publish_sources})
file(GLOB mqtt_sources "${publish}/mqtt/*.cpp")
file(GLOB kafka_sources "${publish}/kafka/*.cpp")
file(GLOB python_sources "${mono}/gst/elements/gvapython/*.cpp" "${mono}/gst/elements/gvapython/*.c")
add_library(gstgvametapublishmqtt MODULE ${mqtt_sources})
add_library(gstgvametapublishkafka MODULE ${kafka_sources})
add_library(gstgvapython MODULE ${python_sources})
find_package(Python3 3.10 REQUIRED COMPONENTS Development)
find_library(PAHO_LIBRARY paho-mqtt3as HINTS "${SYSROOT}/usr/lib/x86_64-linux-gnu" REQUIRED)
find_library(KAFKA_LIBRARY rdkafka++ HINTS "${SYSROOT}/usr/lib/x86_64-linux-gnu" REQUIRED)
find_library(UUID_LIBRARY NAMES uuid libuuid.so.1 REQUIRED)
target_include_directories(gstgvapython PRIVATE ${Python3_INCLUDE_DIRS} "${SYSROOT}/usr/include/pygobject-3.0")
target_link_libraries(gstgvapython PRIVATE Python3::Python)
pkg_check_modules(GMODULE REQUIRED IMPORTED_TARGET gmodule-2.0)
target_link_libraries(gstgvapython PRIVATE PkgConfig::GMODULE)
target_link_libraries(gstgvametapublishmqtt PRIVATE gstgvametapublish ${PAHO_LIBRARY} ${UUID_LIBRARY})
target_link_libraries(gstgvametapublishkafka PRIVATE gstgvametapublish ${KAFKA_LIBRARY})
set_target_properties(gstgvametapublishmqtt gstgvametapublishkafka PROPERTIES INSTALL_RPATH "$ORIGIN")
foreach(target amdva_dlstreamer_common gstdlstreamer_selected gstgvametapublish gstgvametapublishmqtt gstgvametapublishkafka gstgvapython)
  target_include_directories(${target} PRIVATE
    "${CMAKE_BINARY_DIR}" "${DLS_SOURCE}/include" "${DLS_SOURCE}/include/dlstreamer/gst/metadata"
    "${DLS_SOURCE}/include/dlstreamer/gst/videoanalytics" "${DLS_SOURCE}/src/utils" "${mono}/gst/common"
    "${mono}/inference_backend/include" "${track}" "${convert}" "${publish}" "${publish}/base"
    "${SYSROOT}/usr/include/opencv4" "${SYSROOT}/usr/include" "${SYSROOT}/usr/include/librdkafka")
  target_link_directories(${target} PRIVATE "${SYSROOT}/usr/lib/x86_64-linux-gnu")
  target_link_libraries(${target} PRIVATE PkgConfig::GST PkgConfig::VA openvino::runtime opencv_core opencv_imgproc ${CMAKE_DL_LIBS})
  target_compile_definitions(${target} PRIVATE GST_USE_UNSTABLE_API)
  target_compile_options(${target} PRIVATE -include "${CMAKE_BINARY_DIR}/config.h")
  target_link_options(${target} PRIVATE -Wl,--no-undefined)
endforeach()
target_link_libraries(gstdlstreamer_selected PRIVATE amdva_dlstreamer_common)
target_link_libraries(gstgvametapublish PRIVATE amdva_dlstreamer_common)
target_link_libraries(gstgvametapublishmqtt PRIVATE amdva_dlstreamer_common)
target_link_libraries(gstgvametapublishkafka PRIVATE amdva_dlstreamer_common)
target_link_libraries(gstgvapython PRIVATE amdva_dlstreamer_common)
install(TARGETS amdva_dlstreamer_common LIBRARY DESTINATION lib)
install(TARGETS gstdlstreamer_selected gstgvametapublish gstgvametapublishmqtt gstgvametapublishkafka gstgvapython LIBRARY DESTINATION lib/gstreamer-1.0)
install(DIRECTORY "${DLS_SOURCE}/include/dlstreamer/gst/metadata" DESTINATION include/dlstreamer/gst)
install(DIRECTORY "${DLS_SOURCE}/python/gstgva" DESTINATION lib/python3/dist-packages OPTIONAL)
