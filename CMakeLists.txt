cmake_minimum_required(VERSION 3.22)
project(amd_video_analysis VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(VA REQUIRED IMPORTED_TARGET libva libva-drm)
set(AMDVA_ROCM_ROOT "/opt/rocm/core-10.0" CACHE PATH "ROCm runtime root")
set(AMDVA_HIP_INCLUDE "${AMDVA_ROCM_ROOT}/include" CACHE PATH "HIP headers")
find_library(HIP_LIBRARY amdhip64 HINTS "${AMDVA_ROCM_ROOT}/lib" REQUIRED)
find_library(HIPRTC_LIBRARY hiprtc HINTS "${AMDVA_ROCM_ROOT}/lib" REQUIRED)
add_library(amdva_runtime SHARED src/runtime.cpp src/preprocess.cpp src/telemetry.cpp)
target_include_directories(amdva_runtime PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>" "$<BUILD_INTERFACE:${AMDVA_HIP_INCLUDE}>")
target_compile_definitions(amdva_runtime PUBLIC __HIP_PLATFORM_AMD__)
target_link_libraries(amdva_runtime PUBLIC PkgConfig::VA "${HIP_LIBRARY}" "${HIPRTC_LIBRARY}")
target_link_directories(amdva_runtime PRIVATE ${VA_LIBRARY_DIRS})
target_compile_options(amdva_runtime PRIVATE -Wall -Wextra -Wpedantic)
add_executable(amdva-probe tools/probe.cpp)
target_link_libraries(amdva-probe PRIVATE amdva_runtime)
option(AMDVA_BUILD_PLUGINS "Build GStreamer AMD elements" ON)
if(AMDVA_BUILD_PLUGINS)
  add_subdirectory(src/gst)
endif()
find_package(Threads REQUIRED)
pkg_check_modules(AMDGPU REQUIRED IMPORTED_TARGET libdrm_amdgpu)
add_library(amdva_monitor SHARED src/resource_monitor.cpp)
target_include_directories(amdva_monitor PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>")
target_link_libraries(amdva_monitor PRIVATE PkgConfig::AMDGPU)
install(TARGETS amdva_monitor EXPORT AMDVideoAnalysisTargets LIBRARY DESTINATION lib)
option(AMDVA_BUILD_DASHBOARD "Build headless hardware-encoded RTSP dashboard" ON)
if(AMDVA_BUILD_DASHBOARD)
  add_subdirectory(src/dashboard)
endif()
option(AMDVA_BUILD_RTSP_FILES "Build headless file-to-RTSP tool" ON)
if(AMDVA_BUILD_RTSP_FILES)
  add_subdirectory(src/rtsp_files)
endif()
option(AMDVA_BUILD_NVR "Build Qt desktop NVR sample" ON)
option(AMDVA_BUILD_STREAMING "Build headless RTSP, recording and analytics services" ON)
if(AMDVA_BUILD_STREAMING)
  if(NOT AMDVA_BUILD_PLUGINS)
    message(FATAL_ERROR "Streaming services require AMDVA_BUILD_PLUGINS")
  endif()
  add_subdirectory(src/streaming)
endif()
if(AMDVA_BUILD_NVR)
  if(NOT AMDVA_BUILD_STREAMING)
    message(FATAL_ERROR "NVR requires AMDVA_BUILD_STREAMING")
  endif()
  add_subdirectory(src/nvr)
endif()
if(BUILD_TESTING)
  add_executable(resource-monitor-tests tests/resource_monitor_tests.cpp)
  target_link_libraries(resource-monitor-tests PRIVATE amdva_monitor)
  add_test(NAME resource-monitor-contract COMMAND resource-monitor-tests)
  add_executable(telemetry-tests tests/telemetry_tests.cpp)
  target_link_libraries(telemetry-tests PRIVATE amdva_runtime)
  add_test(NAME telemetry-contract COMMAND telemetry-tests)
  add_executable(runtime-tests tests/runtime_tests.cpp)
  target_link_libraries(runtime-tests PRIVATE amdva_runtime)
  add_test(NAME runtime-contract COMMAND runtime-tests)
endif()
install(TARGETS amdva_runtime EXPORT AMDVideoAnalysisTargets LIBRARY DESTINATION lib)
install(TARGETS amdva-probe RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/AMDVideoAnalysisConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/AMDVideoAnalysisConfig.cmake"
  INSTALL_DESTINATION lib/cmake/AMDVideoAnalysis)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/AMDVideoAnalysisConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT AMDVideoAnalysisTargets NAMESPACE AMDVA:: DESTINATION lib/cmake/AMDVideoAnalysis)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/AMDVideoAnalysisConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/AMDVideoAnalysisConfigVersion.cmake" DESTINATION lib/cmake/AMDVideoAnalysis)
