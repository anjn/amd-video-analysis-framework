cmake_minimum_required(VERSION 3.22)
project(amd_video_analysis VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(VA REQUIRED IMPORTED_TARGET libva libva-drm)
set(AMDVA_ROCM_ROOT "/opt/rocm/core-10.0" CACHE PATH "ROCm runtime root")
set(AMDVA_HIP_INCLUDE "${AMDVA_ROCM_ROOT}/include" CACHE PATH "HIP headers")
find_library(HIP_LIBRARY amdhip64 HINTS "${AMDVA_ROCM_ROOT}/lib" REQUIRED)
find_library(HIPRTC_LIBRARY hiprtc HINTS "${AMDVA_ROCM_ROOT}/lib" REQUIRED)
add_library(amdva_runtime SHARED src/runtime.cpp)
target_include_directories(amdva_runtime PUBLIC include "${AMDVA_HIP_INCLUDE}")
target_compile_definitions(amdva_runtime PUBLIC __HIP_PLATFORM_AMD__)
target_link_libraries(amdva_runtime PUBLIC PkgConfig::VA "${HIP_LIBRARY}" "${HIPRTC_LIBRARY}")
target_link_directories(amdva_runtime PUBLIC ${VA_LIBRARY_DIRS})
target_compile_options(amdva_runtime PRIVATE -Wall -Wextra -Wpedantic)
add_executable(amdva-probe tools/probe.cpp)
target_link_libraries(amdva-probe PRIVATE amdva_runtime)
option(AMDVA_BUILD_PLUGINS "Build GStreamer AMD elements" ON)
if(AMDVA_BUILD_PLUGINS)
  add_subdirectory(src/gst)
endif()
option(AMDVA_BUILD_NVR "Build Qt desktop NVR sample" ON)
if(AMDVA_BUILD_NVR)
  if(NOT AMDVA_BUILD_PLUGINS)
    message(FATAL_ERROR "NVR requires AMDVA_BUILD_PLUGINS")
  endif()
  add_subdirectory(src/nvr)
endif()
if(BUILD_TESTING)
  add_executable(runtime-tests tests/runtime_tests.cpp)
  target_link_libraries(runtime-tests PRIVATE amdva_runtime)
  add_test(NAME runtime-contract COMMAND runtime-tests)
endif()
install(TARGETS amdva_runtime amdva-probe LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
