cmake_minimum_required(VERSION 3.22)
project(amd_video_analytics VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
option(AMDVA_HARDWARE_TESTS "Enable tests requiring the validated AMD VA/HIP device" OFF)
find_package(PkgConfig REQUIRED)
include(cmake/AmdvaHostLibraries.cmake)
set(AMDVA_SYSROOT "${CMAKE_SOURCE_DIR}/.deps/sysroot" CACHE PATH "Optional rootless development package sysroot")
option(AMDVA_WITH_MQTT "Build MQTT result publication (Eclipse Paho)" ON)
option(AMDVA_WITH_KAFKA "Build Kafka result publication (librdkafka)" ON)
option(AMDVA_WITH_PYTHON "Build the amdpython element (embedded CPython)" ON)
pkg_check_modules(VA REQUIRED IMPORTED_TARGET libva libva-drm)
set(AMDVA_ROCM_ROOT "/opt/rocm/core-10.0" CACHE PATH "ROCm runtime root")
set(AMDVA_HIP_INCLUDE "${AMDVA_ROCM_ROOT}/include" CACHE PATH "HIP headers")
find_library(HIP_LIBRARY amdhip64 HINTS "${AMDVA_ROCM_ROOT}/lib" REQUIRED)
find_library(HIPRTC_LIBRARY hiprtc HINTS "${AMDVA_ROCM_ROOT}/lib" REQUIRED)
add_library(amdva_runtime SHARED src/runtime.cpp src/preprocess.cpp src/telemetry.cpp)
add_library(AMDVA::amdva_runtime ALIAS amdva_runtime)
target_include_directories(amdva_runtime PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>" "$<BUILD_INTERFACE:${AMDVA_HIP_INCLUDE}>")
target_compile_definitions(amdva_runtime PUBLIC __HIP_PLATFORM_AMD__)
target_link_libraries(amdva_runtime PUBLIC PkgConfig::VA "${HIP_LIBRARY}" "${HIPRTC_LIBRARY}")
target_link_directories(amdva_runtime PRIVATE ${VA_LIBRARY_DIRS})
target_compile_options(amdva_runtime PRIVATE -Wall -Wextra -Wpedantic)
add_executable(amdva-probe tools/probe.cpp)
target_link_libraries(amdva-probe PRIVATE amdva_runtime)
option(AMDVA_BUILD_PLUGINS "Build GStreamer AMD elements" ON)
option(AMDVA_BUILD_GVA_COMPAT "Build vendored DL Streamer gva* elements (gvatrack, gvametaconvert, gvametapublish, gvastreammux)" ON)
if(AMDVA_BUILD_GVA_COMPAT)
  add_subdirectory(third_party/dlstreamer)
endif()
add_subdirectory(src/media)
if(AMDVA_BUILD_PLUGINS)
  add_subdirectory(src/gst)
endif()
find_package(Threads REQUIRED)
pkg_check_modules(AMDGPU REQUIRED IMPORTED_TARGET libdrm_amdgpu)
add_library(amdva_monitor SHARED src/resource_monitor.cpp)
add_library(AMDVA::amdva_monitor ALIAS amdva_monitor)
target_include_directories(amdva_monitor PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>")
target_link_libraries(amdva_monitor PRIVATE PkgConfig::AMDGPU)
install(TARGETS amdva_monitor EXPORT AMDVideoAnalyticsTargets LIBRARY DESTINATION lib)
option(AMDVA_BUILD_STREAMING "Build the RTSP ingest, recording, catalog and playback library" ON)
if(AMDVA_BUILD_STREAMING)
  if(NOT AMDVA_BUILD_PLUGINS)
    message(FATAL_ERROR "Streaming services require AMDVA_BUILD_PLUGINS")
  endif()
  add_subdirectory(src/streaming)
endif()
option(AMDVA_BUILD_EXAMPLES "Build the applications under examples/" ON)
if(AMDVA_BUILD_EXAMPLES)
  add_subdirectory(examples)
endif()
if(BUILD_TESTING)
  add_executable(resource-monitor-tests tests/resource_monitor_tests.cpp)
  target_link_libraries(resource-monitor-tests PRIVATE amdva_monitor)
  add_test(NAME resource-monitor-contract COMMAND resource-monitor-tests)
  add_executable(telemetry-tests tests/telemetry_tests.cpp)
  target_link_libraries(telemetry-tests PRIVATE amdva_runtime)
  add_test(NAME telemetry-contract COMMAND telemetry-tests)
  add_executable(runtime-tests tests/runtime_tests.cpp)
  target_link_libraries(runtime-tests PRIVATE amdva_runtime)
  add_test(NAME runtime-contract COMMAND runtime-tests)
endif()
# 0.x releases promise no ABI compatibility between minor versions.
foreach(library amdva_runtime amdva_monitor amdva_media amdva_inference amdva_gst_support amdva_streaming)
  if(TARGET ${library})
    set_target_properties(${library} PROPERTIES VERSION ${PROJECT_VERSION}
      SOVERSION ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR})
  endif()
endforeach()
install(TARGETS amdva_runtime EXPORT AMDVideoAnalyticsTargets LIBRARY DESTINATION lib)
install(TARGETS amdva-probe RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/AMDVideoAnalyticsConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/AMDVideoAnalyticsConfig.cmake"
  INSTALL_DESTINATION lib/cmake/AMDVideoAnalytics)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/AMDVideoAnalyticsConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT AMDVideoAnalyticsTargets NAMESPACE AMDVA:: DESTINATION lib/cmake/AMDVideoAnalytics)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/AMDVideoAnalyticsConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/AMDVideoAnalyticsConfigVersion.cmake"
  cmake/AmdvaHostLibraries.cmake DESTINATION lib/cmake/AMDVideoAnalytics)
# The package was first published as AMDVideoAnalysis; keep that name working.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/AMDVideoAnalysisConfig.cmake" [=[
# Deprecated: use find_package(AMDVideoAnalytics).
include(CMakeFindDependencyMacro)
find_dependency(AMDVideoAnalytics HINTS "${CMAKE_CURRENT_LIST_DIR}/../AMDVideoAnalytics")
]=])
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/AMDVideoAnalysisConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/AMDVideoAnalysisConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/AMDVideoAnalysisConfigVersion.cmake" DESTINATION lib/cmake/AMDVideoAnalysis)
