# The rootless sysroot contains GLib development archives but no GLib runtime.
# Link the host shared libraries explicitly, avoiding a second GObject runtime
# embedded into the inference library when CMAKE_PREFIX_PATH selects the sysroot.
pkg_get_variable(AMDVA_GLIB_LIBDIR glib-2.0 libdir)
foreach(component gio-2.0 gobject-2.0 glib-2.0)
  find_library(AMDVA_SHARED_${component} NAMES ${component}
    PATHS "${AMDVA_GLIB_LIBDIR}" NO_DEFAULT_PATH REQUIRED)
endforeach()
get_target_property(AMDVA_JSON_LINKS PkgConfig::JSON INTERFACE_LINK_LIBRARIES)
foreach(component gio-2.0 gobject-2.0 glib-2.0)
  list(FILTER AMDVA_JSON_LINKS EXCLUDE REGEX "/lib${component}\\.(a|so)$")
  list(APPEND AMDVA_JSON_LINKS "${AMDVA_SHARED_${component}}")
endforeach()
set_target_properties(PkgConfig::JSON PROPERTIES INTERFACE_LINK_LIBRARIES "${AMDVA_JSON_LINKS}")
