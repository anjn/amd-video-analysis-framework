cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(amdva_rtsp_files LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  find_package(PkgConfig REQUIRED)
  include(CTest)
endif()
pkg_check_modules(FILE_RTSP REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.28.2 gstreamer-app-1.0 gstreamer-rtsp-server-1.0)
add_executable(amdva-rtsp-files main.cpp)
target_link_libraries(amdva-rtsp-files PRIVATE PkgConfig::FILE_RTSP)
target_compile_options(amdva-rtsp-files PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS amdva-rtsp-files RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_test(NAME rtsp-files-help COMMAND amdva-rtsp-files --help)
  add_test(NAME rtsp-files-invalid-port COMMAND amdva-rtsp-files --port 0)
  add_test(NAME rtsp-files-missing-input COMMAND amdva-rtsp-files)
  set_tests_properties(rtsp-files-invalid-port rtsp-files-missing-input PROPERTIES WILL_FAIL TRUE)
endif()
