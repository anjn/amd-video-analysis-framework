cmake_minimum_required(VERSION 3.22)
project(amdva_headless_example LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(AMDVideoAnalysis REQUIRED)
add_executable(telemetry telemetry.cpp)
target_link_libraries(telemetry PRIVATE AMDVA::amdva_gst_support)
add_executable(playback playback.cpp)
target_link_libraries(playback PRIVATE AMDVA::amdva_streaming)
add_executable(timeline timeline.cpp)
target_link_libraries(timeline PRIVATE AMDVA::amdva_streaming)
add_executable(keyframe-preview keyframe-preview.cpp)
target_link_libraries(keyframe-preview PRIVATE AMDVA::amdva_streaming)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDK_APP REQUIRED IMPORTED_TARGET gstreamer-app-1.0 gstreamer-rtsp-server-1.0 glib-2.0)
add_executable(transcode transcode.cpp)
target_link_libraries(transcode PRIVATE AMDVA::amdva_media PkgConfig::SDK_APP)
