cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(amdva_resource_dashboard LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  include(CTest)
  find_package(AMDVideoAnalysis REQUIRED)
endif()
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(DASH_GST REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.28.2 gstreamer-video-1.0 gstreamer-va-1.0 gstreamer-app-1.0 gstreamer-rtsp-server-1.0 gstreamer-codecparsers-1.0)
pkg_check_modules(DASH_GL REQUIRED IMPORTED_TARGET egl glesv2)
pkg_check_modules(DASH_FONT REQUIRED IMPORTED_TARGET freetype2 fontconfig)
add_executable(amdva-dashboard main.cpp renderer.cpp)
target_compile_definitions(amdva-dashboard PRIVATE GST_USE_UNSTABLE_API)
target_link_libraries(amdva-dashboard PRIVATE AMDVA::amdva_media AMDVA::amdva_runtime AMDVA::amdva_monitor PkgConfig::DASH_GST PkgConfig::DASH_GL PkgConfig::DASH_FONT Threads::Threads ${CMAKE_DL_LIBS})
install(TARGETS amdva-dashboard RUNTIME DESTINATION bin)

if(BUILD_TESTING)
  add_test(NAME dashboard-help COMMAND amdva-dashboard --help)
  set_tests_properties(dashboard-help PROPERTIES PASS_REGULAR_EXPRESSION "power-max-w")
  add_test(NAME dashboard-invalid-fps COMMAND amdva-dashboard --fps 0)
  add_test(NAME dashboard-invalid-power COMMAND amdva-dashboard --power-max-w nan)
  add_test(NAME dashboard-invalid-mount COMMAND amdva-dashboard --mount invalid)
  set_tests_properties(dashboard-invalid-fps dashboard-invalid-power dashboard-invalid-mount PROPERTIES WILL_FAIL TRUE)
endif()
