cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(amdva_nvr LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  include(CTest)
  find_package(AMDVideoAnalytics REQUIRED)
  option(AMDVA_BUILD_NVR_GUI "Build the Qt desktop NVR" ON)
  option(AMDVA_HARDWARE_TESTS "Enable tests requiring the validated AMD VA/HIP device" OFF)
endif()
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
find_package(PNG REQUIRED)
pkg_check_modules(APP REQUIRED IMPORTED_TARGET gstreamer-app-1.0)
pkg_check_modules(RTSP_SERVER REQUIRED IMPORTED_TARGET gstreamer-rtsp-server-1.0)
pkg_check_modules(NVR_SQLITE REQUIRED IMPORTED_TARGET sqlite3)
pkg_check_modules(NVR_JSON REQUIRED IMPORTED_TARGET json-glib-1.0)
amdva_use_host_glib(PkgConfig::NVR_JSON)
amdva_use_host_sqlite(PkgConfig::NVR_SQLITE)
set(NVR_SQLITE_VEC "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/sqlite-vec")

# Qt-free backend: camera controller, configuration, AI scheduling and person search.
add_library(amdva_nvr_backend SHARED
  backend/controller.cpp backend/config.cpp backend/ai.cpp backend/search.cpp
  backend/search_index.cpp backend/person_tracking.cpp backend/reid.cpp
  "${NVR_SQLITE_VEC}/sqlite-vec.c")
set_source_files_properties("${NVR_SQLITE_VEC}/sqlite-vec.c" PROPERTIES COMPILE_DEFINITIONS "SQLITE_CORE;SQLITE_VEC_STATIC;SQLITE_VEC_OMIT_FS")
target_include_directories(amdva_nvr_backend PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE "${NVR_SQLITE_VEC}")
target_link_libraries(amdva_nvr_backend
  PUBLIC AMDVA::amdva_streaming AMDVA::amdva_gst_support AMDVA::amdva_inference AMDVA::amdva_runtime
  PRIVATE PkgConfig::NVR_JSON PkgConfig::APP PkgConfig::NVR_SQLITE PNG::PNG Threads::Threads ${CMAKE_DL_LIBS})
install(TARGETS amdva_nvr_backend LIBRARY DESTINATION lib)
install(FILES "${NVR_SQLITE_VEC}/LICENSE" DESTINATION share/licenses/amdva/sqlite-vec)
# Native worker intentionally links neither HIP nor the framework: NPU owns
# its runtime in a clean process, isolated from the NVR's GPU initialization.
add_executable(amdva-mobileclip-worker backend/mobileclip_worker.cpp)
target_include_directories(amdva-mobileclip-worker PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(amdva-mobileclip-worker PRIVATE ${CMAKE_DL_LIBS})
install(TARGETS amdva-mobileclip-worker RUNTIME DESTINATION bin)
target_compile_definitions(amdva_nvr_backend PRIVATE AMDVA_MOBILECLIP_WORKER="$<TARGET_FILE:amdva-mobileclip-worker>")
add_dependencies(amdva_nvr_backend amdva-mobileclip-worker)

add_executable(amdva-streams headless.cpp)
target_link_libraries(amdva-streams PRIVATE amdva_nvr_backend)
install(TARGETS amdva-streams RUNTIME DESTINATION bin)
add_executable(amdva-nvr-rtsp-server tools/nvr_rtsp_server.cpp)
target_link_libraries(amdva-nvr-rtsp-server PRIVATE PkgConfig::RTSP_SERVER PkgConfig::APP)

if(BUILD_TESTING)
  add_executable(search-tests tests/search_tests.cpp)
  target_link_libraries(search-tests PRIVATE amdva_nvr_backend PkgConfig::NVR_SQLITE)
  add_test(NAME search-contract COMMAND search-tests)
  add_executable(ai-tests tests/ai_tests.cpp)
  target_link_libraries(ai-tests PRIVATE amdva_nvr_backend)
  add_test(NAME ai-runtime-contract COMMAND ai-tests)
  set_tests_properties(ai-runtime-contract PROPERTIES TIMEOUT 20)
endif()
if(BUILD_TESTING AND AMDVA_HARDWARE_TESTS)
  add_executable(amdva-reid-validate tools/reid_validate.cpp)
  target_link_libraries(amdva-reid-validate PRIVATE amdva_nvr_backend)
  add_executable(amdva-ai-benchmark tools/ai_benchmark.cpp)
  target_link_libraries(amdva-ai-benchmark PRIVATE AMDVA::amdva_inference)
  add_executable(amdva-ai-infer tools/ai_infer.cpp)
  target_link_libraries(amdva-ai-infer PRIVATE AMDVA::amdva_inference)
  add_executable(amdva-batch-reference tools/batch_reference.cpp)
  target_link_libraries(amdva-batch-reference PRIVATE AMDVA::amdva_inference)
endif()

if(NOT AMDVA_BUILD_NVR_GUI)
  return()
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets OpenGL)
pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl glesv2)
add_library(amdva_nvr_ui STATIC app/window.cpp app/vms.cpp app/search_page.cpp app/ai_page.cpp app/timeline.cpp app/vector_text.cpp)
target_include_directories(amdva_nvr_ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/app)
target_link_libraries(amdva_nvr_ui PUBLIC amdva_nvr_backend AMDVA::amdva_media Qt6::Widgets Qt6::OpenGL PkgConfig::EGL)
add_executable(amdva-nvr app/main.cpp)
target_link_libraries(amdva-nvr PRIVATE amdva_nvr_ui)
install(TARGETS amdva-nvr RUNTIME DESTINATION bin)

  add_executable(search-conditions-tests tests/search_conditions_tests.cpp)

  add_test(NAME search-conditions-contract COMMAND search-conditions-tests)
  set_tests_properties(search-conditions-contract PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

  # Display, camera and model validation programs; see README.md and docs/.
  add_executable(nvr-renderer-tests tests/nvr_renderer_tests.cpp)
  target_link_libraries(nvr-renderer-tests PRIVATE amdva_nvr_ui)
  add_executable(nvr-timing-tests tests/nvr_timing_tests.cpp)
  target_link_libraries(nvr-timing-tests PRIVATE amdva_nvr_ui)
  add_executable(amdva-nvr-validate tools/nvr_validate.cpp)
  target_link_libraries(amdva-nvr-validate PRIVATE amdva_nvr_ui ${CMAKE_DL_LIBS})
  add_executable(amdva-nvr-faults tools/nvr_faults.cpp)
  target_link_libraries(amdva-nvr-faults PRIVATE amdva_nvr_ui)

  target_link_libraries(amdva-dashboard-nvr-validate PRIVATE amdva_nvr_ui)

if(BUILD_TESTING AND AMDVA_HARDWARE_TESTS)
  add_executable(amdva-ai-validate tools/ai_validate.cpp)
